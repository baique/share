# @baique/share 局域网共享终端 —— 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现 `npx @baique/share` 启动的局域网聊天室式共享终端：mDNS 自动发现、文本/文件广播收发、30 分钟历史同步、按序号下载文件。

**Architecture:** 无中心 P2P。每台机器既是服务器（TCP 收消息 + HTTP 供文件下载 + mDNS 广播身份）也是客户端（mDNS 发现 + TCP 推送 + HTTP 拉文件）。消息推送采用"发送时临时建连，发完即断"模型，不做长连接。历史仅内存保留 30 分钟，新节点通过 sync 协议拉取增量。

**Tech Stack:** Node.js ≥18（ESM）、Ink@5 + React@18（TUI）、ink-text-input@5（输入框）、multicast-dns@7（mDNS）、Node 原生 net/http/crypto/fs/os。测试用内置 `node:test`。

## Global Constraints

- Node ≥ 18，ESM（`"type": "module"`），所有模块 `import`/`export`
- 依赖仅限：`ink`、`react`、`ink-text-input`、`multicast-dns`；其余一律用 Node 内置模块
- 无任何配置文件、无设置项；唯一 CLI 参数 `-c <ip>`
- 消息协议：JSON 对象 + `\n` 分隔，走 TCP；字段命名全局统一（见 Task 2/3 定义）
- 身份 = 主机名（`os.hostname()`），节点 id = `crypto.randomUUID()`
- 消息保留 30 分钟，仅内存；文件流式传输不进内存
- 测试用 `node:test`，测试数据用 `try...finally` 清理，不保留
- 每次任务结束必须 git commit

import { setTimeout as delay } from 'node:timers/promises';

export function createHistory({ ttlMs = 30 * 60 * 1000, now = Date.now } = {}) {
  const messages = [];
  const prune = () => {
    const cutoff = now() - ttlMs;
    while (messages.length && messages[0].time < cutoff) messages.shift();
  };
  return {
    add(msg) {
      messages.push(msg);
      prune();
    },
    getAll() {
      prune();
      return messages.slice();
    },
    prune,
    // 供 index.js 定时兜底清理
    _timer: null,
  };
}
```

export function createNodeList({ ttlMs = 60 * 1000, now = Date.now } = {}) {
  const nodes = new Map();
  return {
    upsert(node) {
      const isNew = !nodes.has(node.id);
      nodes.set(node.id, { ...node, lastSeen: now() });
      return isNew;
    },
    get(id) { return nodes.get(id); },
    remove(id) { nodes.delete(id); },
    prune() {
      const cutoff = now() - ttlMs;
      for (const [id, n] of nodes) if (n.lastSeen < cutoff) nodes.delete(id);
    },
    all() { return [...nodes.values()]; },
  };
}
```

### Task 4: 网络服务 server.js（TCP 接收 + HTTP 文件）

**Files:**
- Create: `src/server.js`

**Interfaces:**
- Produces: `createServer({ onMessage, onSync })` → `{ tcpPort, httpPort, close(), registerFile(id, info) }`
  - 内部自动监听 127.0.0.1 + 本机网卡 IP 上的 TCP（起始 41023，占用则 +1）与 HTTP（起始 41024，占用则 +1）
  - TCP 收 `\n` 分隔 JSON 行：

  - `close()`: 关闭全部监听

```js
import net from 'node:net';
import http from 'node:http';
import { createReadStream } from 'node:fs';
import os from 'node:os';
import { networkInterfaces } from 'node:os';

const LINES = /\r?\n/;

function localIPv4() {
  for (const ifs of Object.values(networkInterfaces())) {
    for (const iface of ifs || []) {
      if (iface.family === 'IPv4' && !iface.internal) return iface.address;
    }
  }
  return '127.0.0.1';
}

function listen(server, startPort) {
  return new Promise((resolve) => {
    const tryPort = (p) => {
      server.once('error', (err) => {
        if (err.code === 'EADDRINUSE') tryPort(p + 1);
        else throw err;
      });
      server.listen(p, () => resolve(server.address().port));
    };
    tryPort(startPort);
  });
}

export async function createServer({ onMessage, onSync }) {
  const host = localIPv4();
  const files = new Map();

  // --- TCP ---
  const tcp = net.createServer((sock) => {
    sock.setEncoding('utf8');
    let buf = '';
    sock.on('data', (chunk) => {
      buf += chunk;
      let idx;
      while ((idx = buf.indexOf('\n')) !== -1) {
        const raw = buf.slice(0, idx); buf = buf.slice(idx + 1);
        if (!raw.trim()) continue;
        let msg;
        try { msg = JSON.parse(raw); } catch { continue; }
        if (msg.type === 'sync') {
          const history = onSync({ id: msg.id, name: msg.name });
          sock.write(JSON.stringify({
            type: 'welcome', id: os.hostname(), name: os.hostname(),
            port: tcpPort, httpPort,
          }) + '\n');
          sock.write(JSON.stringify({ type: 'sync-response', id: os.hostname(), name: os.hostname(), messages: history }) + '\n');
        } else if (msg.type === 'text' || msg.type === 'file') {
          onMessage(msg);
        }
      }
    });
    sock.on('end', () => sock.end());
    sock.on('error', () => {});
  });

  // --- HTTP ---
  const httpServer = http.createServer((req, res) => {
    const m = /^\/f\/([^/]+)$/.exec(req.url || '');
    if (!m || !files.has(m[1])) { res.writeHead(404); res.end('not found'); return; }
    const { path, name } = files.get(m[1]);
    res.writeHead(200, {
      'Content-Disposition': `attachment; filename="${encodeURIComponent(name)}"`,
    });
    createReadStream(path).pipe(res);
  });

  await new Promise((r) => tcp.listen ? null : null, r);
  const tcpPort = await listen(tcp, 41023);
  const httpPort = await listen(httpServer, 41024);

  return {
    tcpPort,
    httpPort,
    registerFile(id, info) { files.set(id, info); },
    close() {
      return Promise.all([
        new Promise(r => tcp.close(r)),
        new Promise(r => httpServer.close(r)),
      ]);
    },
  };
}
```

> 注意：上面对 `await new Promise(...)` 有一行冗余写法，实现时删除，直接用 `const tcpPort = await listen(...)`。

### Task 5: mDNS 发现 discovery.js

**Files:**
- Create: `src/discovery.js`

**Interfaces:**
- Consumes: 无（依赖 multicast-dns）
- Produces: `createDiscovery({ id, name, port, httpPort }, { onNode })` → `{ start(), stop() }`
  - `onNode(node)`: 每收到一条 mDNS 响应，回调 `{ id, name, ip, port, httpPort }`（供 node-list.upsert）
  - 内部：加入 224.0.0.251:5353 组播，响应 `_share._tcp.local` 查询，每 10s 主动 announce + query，收到响应回调；`stop()` 发送 TTL=0 goodbye 并销毁
  - 同机多实例：multicast-dns 已启用 reuseAddr，多实例可共存

- [ ] **Step 1: 实现 discovery.js**

```js
import multicastdns from 'multicast-dns';
import os from 'node:os';

const SERVICE = '_share._tcp.local';

export function createDiscovery({ id, name, port, httpPort }, { onNode }) {
  const mdns = multicastdns({ multicast: true, interface: '0.0.0.0', port: 5353, reuseAddr: true });

  const answerFor = () => [{
    name: SERVICE, type: 'PTR', ttl: 60, data: `${id}.${SERVICE}`,
  }, {
    name: `${id}.${SERVICE}`, type: 'SRV', ttl: 60,
    data: { port, target: `${id}.local.` },
  }, {
    name: `${id}.${SERVICE}`, type: 'TXT', ttl: 60,
    data: [Buffer.from(`id=${id}`), Buffer.from(`name=${name}`), Buffer.from(`http=${httpPort}`)],
  }, {
    name: `${id}.local.`, type: 'A', ttl: 60, data: os.networkInterfaces 已由系统决定
  }];

  mdns.on('query', (q, rinfo) => {
    const hit = q.questions && q.questions.some(x => x.name === SERVICE);
    if (!hit) return;
    mdns.respond(answerFor());
  });

  mdns.on('response', (res, rinfo) => {
    const txt = res.answers.find(a => a.type === 'TXT' && a.name.endsWith(SERVICE));
    if (!txt) return;
    const data = Object.fromEntries(txt.data.map(b => {
      const s = b.toString(); const i = s.indexOf('='); return [s.slice(0, i), s.slice(i + 1)];
    }));
    if (!data.id || data.id === id) return;
    onNode({ id: data.id, name: data.name || data.id, ip: rinfo.address, port: data.port ? Number(data.port) : 0, httpPort: data.http ? Number(data.http) : 0 });
  });

  const tick = () => { mdns.respond(answerFor()); mdns.query({ questions: [{ name: SERVICE, type: 'PTR' }] }); };
  const timer = setInterval(tick, 10_000);

  return {
    start() { tick(); },
    stop() {
      clearInterval(timer);
      try {
        mdns.respond([{
          name: `${id}.${SERVICE}`, type: 'SRV', ttl: 0, data: { port, target: `${id}.local.` },
        }]);
      } catch {}
      mdns.destroy();
    },
  };
}
```

> 说明：SRV `data.port` 与 TXT `port` 字段重复，为避免 SRV 解析歧义，TXT 中 `port` 为 TCP 端口、`http` 为 HTTP 端口；`onNode` 只信任 `rinfo.address`（发送方源 IP）。

```js
import net from 'node:net';
import { stat } from 'node:fs/promises';
import path from 'node:path';
import os from 'node:os';

function oneShot(ip, port, payload) {
  return new Promise((resolve) => {
    const sock = net.createConnection({ host: ip, port });
    let buf = '';
    sock.setEncoding('utf8');
    const done = (ok) => { try { sock.destroy(); } catch {} resolve(ok); };
    sock.on('connect', () => sock.write(JSON.stringify(payload) + '\n'));
    sock.on('data', d => { buf += d; });
    sock.on('end', () => done(true));
    sock.on('error', () => done(false));
    setTimeout(() => done(false), 2000);
  });
}

export async function pushMessage(node, msg) {
  return oneShot(node.ip, node.port, msg);
}

export async function requestSync(node, selfId, selfName, seenIds = []) {
  const ip = node.ip, port = node.port;
  return new Promise((resolve) => {
    const sock = net.createConnection({ host: ip, port });
    let buf = '';
    let nodeInfo = null, messages = null;
    sock.setEncoding('utf8');
    sock.on('connect', () => sock.write(JSON.stringify({ type: 'sync', id: selfId, name: selfName, seenIds }) + '\n'));
    sock.on('data', d => {
      buf += d;
      let idx;
      while ((idx = buf.indexOf('\n')) !== -1) {
        const raw = buf.slice(0, idx); buf = buf.slice(idx + 1);
        if (!raw.trim()) continue;
        let msg; try { msg = JSON.parse(raw); } catch { continue; }
        if (msg.type === 'welcome') nodeInfo = { id: msg.id, name: msg.name, ip, port: msg.port, httpPort: msg.httpPort };
        if (msg.type === 'sync-response') messages = msg.messages;
      }
      if (nodeInfo && messages) { sock.end(); }
    });
    sock.on('end', () => resolve(nodeInfo && messages ? { node: nodeInfo, messages } : null));
    sock.on('error', () => resolve(null));
    setTimeout(() => resolve(null), 2000);
  });
}

export function buildFileInfo(server, fileId, filePath) {
  // 同步 stat 以在 UI 线程直接 throw；文件不存在时抛错由调用方处理
  const st = require('node:fs').statSync(filePath);
  if (!st.isFile()) throw new Error('not a file: ' + filePath);
  server.registerFile(fileId, { path: filePath, name: path.basename(filePath), size: st.size });
  return { name: path.basename(filePath), size: st.size, url: `http://${os.hostname() 占位}/${server.httpPort}/f/${fileId}` };
}
```

> 修正：`buildFileInfo` 的 url 应为本机可达 IP（与 server 监听地址一致），实现时用 `server.ip || localIPv4()`；Task 4 的 server 需导出 `ip` 字段。

- `q` 退出；`↑/↓` 或 `j/k` 滚动；数字键 `1-9` 触发对应序号下载

import React, { useEffect, useMemo, useState } from 'react';
import { Box, Text, useInput, useApp } from 'ink';
import TextInput from 'ink-text-input';
import { existsSync } from 'node:fs';

const fmtTime = (t) => {
  const d = new Date(t);
  return `${String(d.getHours()).padStart(2, '0')}:${String(d.getMinutes()).padStart(2, '0')}`;
};
const fmtSize = (n) => n >= 1048576 ? (n / 1048576).toFixed(1) + 'MB' : n >= 1024 ? (n / 1024).toFixed(0) + 'KB' : n + 'B';

function Messages({ messages, fileSeq, status }) {
  const rows = messages.map(m => {
    if (m.type === 'file') {
      const seq = fileSeq.get(m.id);
      return `[${fmtTime(m.time)}] ${m.from}: [文件] ${m.file.name}(${fmtSize(m.file.size)})  ${seq ? `[下载 ${seq}]` : ''}`;
    }
    return `[${fmtTime(m.time)}] ${m.from}: ${m.text}`;
  });
  const visible = rows.slice(-200);
  return <Box flexDirection="column">
    {visible.map((r, i) => <Text key={i}>{r}</Text>)}
    {status.map((s, i) => <Text key={'s' + i} color="cyan">{s}</Text>)}
  </Box>;
}

function App({ runtime }) {
  const { useApp: _ua } = { useApp };
  const [input, setInput] = useState('');
  const [scroll, setScroll] = useState(0);
  const [status, setStatus] = useState([]);
  const fileSeq = useMemo(() => {
    const m = new Map(); let n = 0;
    runtime.messages.forEach(x => { if (x.type === 'file') m.set(x.id, ++n); });
    return m;
  }, [runtime.messages]);

  useInput((ch, key) => {
    if (key.ctrl && ch === 'c') { runtime.quit(); process.exit(0); }
    if (ch === 'q') { runtime.quit(); process.exit(0); }
    if (key.upArrow || ch === 'k') setScroll(s => Math.min(s + 1, runtime.messages.length - 1));
    if (key.downArrow || ch === 'j') setScroll(s => Math.max(s - 1, 0));
    if (/^[1-9]$/.test(ch)) {
      const target = [...fileSeq.entries()].find(([, n]) => n === Number(ch));
      if (target) {
        const fileMsg = runtime.messages.find(x => x.id === target[0]);
        runtime.download(fileMsg).then(() => {
          setStatus(s => [...s, `下载完成 → ./${fileMsg.file.name}`]);
        }).catch(e => {
          setStatus(s => [...s, `下载失败: ${e.message}`]);
        });
      }
    }
  });

  const onSubmit = async (value) => {
    const trimmed = value.trim();
    if (!trimmed) return;
    setInput('');
    if (existsSync(trimmed)) {
      const r = await runtime.sendFile(trimmed);
      setStatus(s => [...s, r.ok ? `已发送文件 ${trimmed}` : `发送失败: ${r.msg}`]);
    } else {
      await runtime.sendText(trimmed);
    }
  };

  return <Box flexDirection="column">
    <Box><Text bold color="green">LAN Share — {runtime.nodes.length} 台设备在线</Text></Box>
    <Box flexDirection="column" marginTop={1}>
      <Messages messages={runtime.messages} fileSeq={fileSeq} status={status} />
    </Box>
    <Box marginTop={1}><Text color="gray">输入文本或文件路径回车发送 · ↑/↓滚动 · 数字下载 · q退出</Text></Box>
    <Box><Text color="gray">› </Text><TextInput value={input} onChange={setInput} onSubmit={onSubmit} /></Box>
  </Box>;
}

export function renderUI(runtime) {
  const ink = require('ink');
  return ink.render(React.createElement(App, { runtime }));
}

```js
import os from 'node:os';
import crypto from 'node:crypto';
import { createHistory } from './history.js';
import { createNodeList } from './node-list.js';
import { createServer } from './server.js';
import { createDiscovery } from './discovery.js';
import { pushMessage, requestSync, buildFileInfo } from './send.js';
import { renderUI } from './ui.js';

export async function main(argv) {
  const selfId = crypto.randomUUID().slice(0, 8);
  const selfName = os.hostname();

  const history = createHistory();
  const nodes = createNodeList();
  let lastFileSeq = 0;
  const fileSeq = new Map();

  const server = await createServer({
    onMessage(msg) {
      history.add(msg);
      if (msg.type === 'file') { fileSeq.set(msg.id, ++lastFileSeq); }
      refresh();
    },
    onSync() { return history.getAll(); },
  });

  const discovery = createDiscovery({ id: selfId, name: selfName, port: server.tcpPort, httpPort: server.httpPort }, {
    onNode(node) {
      if (nodes.upsert(node)) {
        refresh();
        // 新节点出现，拉历史
        requestSync(node, selfId, selfName, []).then(r => {
          if (!r) return;
          nodes.upsert(r.node);
          for (const m of r.messages) { if (!history.getAll().some(x => x.id === m.id)) history.add(m); }
          refresh();
        });
      }
    },
  });
  discovery.start();

  // -c 直连
  const ci = argv.indexOf('-c');
  if (ci !== -1 && argv[ci + 1]) {
    const spec = argv[ci + 1];
    const [ip, port] = spec.split(':');
    const node = { id: spec, name: ip, ip, port: Number(port) || 41023 };
    requestSync(node, selfId, selfName, []).then(r => {
      if (!r) { console.log(`无法连接 ${spec}`); return; }
      nodes.upsert(r.node);
      for (const m of r.messages) { if (!history.getAll().some(x => x.id === m.id)) history.add(m); }
      refresh();
    });
  }

  // 周期清理 + 刷新
  setInterval(() => { nodes.prune(); refresh(); }, 10_000);
  const pruneTimer = setInterval(() => history.prune(), 60_000);

  function refresh() { /* 由 ui 轮询驱动，见 Step 2 */ }

  const ui = renderUI({
    messages: history.getAll(),
    nodes: nodes.all(),
    onlineCount: () => nodes.all().length,
    sendText(text) {
      const msg = { id: crypto.randomUUID(), type: 'text', from: selfName, time: Date.now(), text };
      history.add(msg); refresh();
      for (const n of nodes.all()) pushMessage(n, msg);
    },
    sendFile(filePath) {
      try {
        const fileId = crypto.randomUUID().slice(0, 8);
        const info = buildFileInfo(server, fileId, filePath);
        const msg = { id: crypto.randomUUID(), type: 'file', from: selfName, time: Date.now(), file: info };
        history.add(msg); fileSeq.set(msg.id, ++lastFileSeq); refresh();
        for (const n of nodes.all()) pushMessage(n, msg);
        return { ok: true };
      } catch (e) {
        return { ok: false, msg: e.message };
      }
    },
    async download(fileMsg) {
      const res = await fetch(fileMsg.file.url);
      if (!res.ok) throw new Error(`HTTP ${res.status}`);
      const buf = Buffer.from(await res.arrayBuffer());
      const { writeFile } = await import('node:fs/promises');
      const { fileURLToPath } = await import('node:url');
      let name = fileMsg.file.name;
      const { existsSync } = await import('node:fs');
      let target = name;
      let i = 1;
      while (existsSync(target)) { target = `${basename_noext(name)} (${i++})${ext(name)}`; }
      await writeFile(target, buf);
    },
    quit() {
      discovery.stop();
      server.close();
      clearInterval(pruneTimer);
      process.exit(0);
    },
  });
  return ui;
}

function basename_noext(n) { const i = n.lastIndexOf('.'); return i > 0 ? n.slice(0, i) : n; }
function ext(n) { const i = n.lastIndexOf('.'); return i > 0 ? n.slice(i) : ''; }
```

> 说明：`refresh` 目前为空函数——ui 轮询驱动在 Task 9 中改为事件驱动；`basename_noext/ext` 用于文件名冲突加后缀。
